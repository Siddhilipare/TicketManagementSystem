using System;

using ListDal;

using ListDal.Dal;

namespace List.Controllers
{
    public class MovieController : Controller
    {
        public ActionResult MovieList( )
        {
            MovieDal moviedal = new MovieDal();
            var movieList = moviedal.GetMovieList();

            return View(movieList);
        }

        public ActionResult Create()
        {

            return View();
        }

        /*18 May Admin->Create movie */
        [HttpPost]
        public ActionResult Create(MovieViewModel movie)
        {
            try
            {

                movie.CreatedBy = 1;
                movie.LastModifiedBy = 1;
                movie.IsActive = true;


                MovieDal moviedal = new MovieDal();

                int result = moviedal.InsertMovie(movie);

                if (result > 0)
                {
                    return RedirectToAction("MovieList");
                }

                return View(movie);
            }

            catch
            {
                throw;
            }
        }

        /*19 May Edit controller*/

        
        public ActionResult EditMovie(int movieID)
        {
            MovieDal moviedal = new MovieDal();
            MovieViewModel movie = moviedal.GetMovieById(movieID);

            return View(movie);
        }
        [HttpPost]
        public ActionResult EditMovie(MovieViewModel movie)
        {
            movie.LastModifiedBy = 1;

            MovieDal moviedal = new MovieDal();

            int result = moviedal.UpdateMovie(movie);

            if (result > 0)
            {
                return RedirectToAction("MovieList");
            }

            return View(movie);
        }

        /*Delete movie controller*/
        [HttpPost]
        public ActionResult DeleteMovie(int movieID)
        {
            MovieDal moviedal = new MovieDal();
            MovieViewModel movie = moviedal.GetMovieById(movieID);

            return View(movie);
        }
    }
}

using System;
using System.Collections.Generic;
using System.Data;
using System.Data.Common;
using Microsoft.Practices.EnterpriseLibrary.Data;
using ListModel.ViewModel;

namespace ListDal.Dal
{
    public class MovieDal
    {
        private Database db;

        public MovieDal()
        {
            this.db = DatabaseFactory.CreateDatabase();
        }

        public List<MovieViewModel> GetMovieList()
        {
            List<MovieViewModel> movieList = new List<MovieViewModel>();

            try
            {
                DbCommand dbCommand = db.GetStoredProcCommand("GetMovieList");

                DataSet ds = db.ExecuteDataSet(dbCommand);

                if (ds.Tables.Count > 0 && ds.Tables[0].Rows.Count > 0)
                {
                    foreach (DataRow row in ds.Tables[0].Rows)
                    {
                        MovieViewModel movie = new MovieViewModel
                        {
                            MovieID = row["MovieID"] != DBNull.Value
         ? Convert.ToInt32(row["MovieID"])
         : 0,

                            MovieName = row["MovieName"] != DBNull.Value
         ? row["MovieName"].ToString()
         : "",

                            Description = row["Description"] != DBNull.Value
         ? row["Description"].ToString()
         : "",

                            MovieImage = row["MovieImage"] != DBNull.Value
         ? row["MovieImage"].ToString()
         : "",

                            Duration = row["Duration"] != DBNull.Value
         ? Convert.ToInt32(row["Duration"])
         : 0,

                            Language = row["Language"] != DBNull.Value
         ? row["Language"].ToString()
         : "",

                            Genre = row["Genre"] != DBNull.Value
         ? row["Genre"].ToString()
         : "",

                            ReleaseDate = row["ReleaseDate"] != DBNull.Value
         ? Convert.ToDateTime(row["ReleaseDate"])
         : DateTime.MinValue,

                            IsActive = row["IsActive"] != DBNull.Value
         ? Convert.ToBoolean(row["IsActive"])
         : false,

                            CreatedAt = row["CreatedAt"] != DBNull.Value
         ? Convert.ToDateTime(row["CreatedAt"])
         : DateTime.MinValue,

                            CreatedBy = row["CreatedBy"] != DBNull.Value
         ? Convert.ToInt32(row["CreatedBy"])
         : 0,

                            LastModifiedAt = row["LastModifiedAt"] != DBNull.Value
         ? Convert.ToDateTime(row["LastModifiedAt"])
         : DateTime.MinValue,

                            LastModifiedBy = row["LastModifiedBy"] != DBNull.Value
         ? Convert.ToInt32(row["LastModifiedBy"])
         : 0
                        };

                        movieList.Add(movie);
                    }
                }
            }
            catch
            {
                throw;
            }

            return movieList;
        }

        /*18 May Admin->Create movie */
        public int InsertMovie(MovieViewModel movie)
        {
            int resultIsInserted = 0;

            try
            {

                DbCommand com = db.GetStoredProcCommand("InsertMovieinDb");

                db.AddInParameter(com, "@MovieName", DbType.String,
                    !string.IsNullOrEmpty(movie.MovieName)
                                               ? (object)movie.MovieName
                                               : DBNull.Value);
                db.AddInParameter(com, "@Description", DbType.String, movie.Description);

                db.AddInParameter(com, "@MovieImage", DbType.String, movie.MovieImage);
                db.AddInParameter(com, "@Duration", DbType.Int32, movie.Duration);
                db.AddInParameter(com, "@Language", DbType.String, movie.Language);
                db.AddInParameter(com, "@Genre", DbType.String, movie.Genre);
                db.AddInParameter(com, "@ReleaseDate", DbType.DateTime, movie.ReleaseDate);
                db.AddInParameter(com, "@IsActive", DbType.Boolean, movie.IsActive);
                db.AddInParameter(com, "@CreatedBy", DbType.Int32, movie.CreatedBy);
                db.AddInParameter(com, "@LastModifiedBy", DbType.Int32, movie.LastModifiedBy);


                resultIsInserted = db.ExecuteNonQuery(com);
            }
            catch(Exception ex)
            {
                throw;
            }
            return resultIsInserted;
        }

        /*19MAY Getting moviedetails through Id*/

        public MovieViewModel GetMovieById(int MovieId)
        {
            MovieViewModel movie = new MovieViewModel();

            try
            {
                DbCommand com = db.GetStoredProcCommand("GetMovieById");

                db.AddInParameter(com, "@MovieId", DbType.Int32, MovieId);

                DataSet ds = db.ExecuteDataSet(com);
                if (ds.Tables.Count > 0 && ds.Tables[0].Rows.Count > 0)
                {
                    DataRow row = ds.Tables[0].Rows[0];

                    movie.MovieID = row["MovieID"] != DBNull.Value
         ? Convert.ToInt32(row["MovieID"])
         : 0;

                    movie.MovieName = row["MovieName"] != DBNull.Value
        ? row["MovieName"].ToString()
        : "";

                    movie.Description = row["Description"] != DBNull.Value
         ? row["Description"].ToString()
         : "";

                    movie.MovieImage = row["MovieImage"] != DBNull.Value
 ? row["MovieImage"].ToString()
 : "";

                    movie.Duration = row["Duration"] != DBNull.Value
  ? Convert.ToInt32(row["Duration"])
  : 0;

                    movie.Language = row["Language"] != DBNull.Value
 ? row["Language"].ToString()
 : "";

                    movie.Genre = row["Genre"] != DBNull.Value
  ? row["Genre"].ToString()
  : "";

                    movie.ReleaseDate = row["ReleaseDate"] != DBNull.Value
  ? Convert.ToDateTime(row["ReleaseDate"])
  : DateTime.MinValue;

                    movie.IsActive = row["IsActive"] != DBNull.Value
  ? Convert.ToBoolean(row["IsActive"])
  : false;

                    movie.CreatedAt = row["CreatedAt"] != DBNull.Value
  ? Convert.ToDateTime(row["CreatedAt"])
  : DateTime.MinValue;

                    movie.CreatedBy = row["CreatedBy"] != DBNull.Value
  ? Convert.ToInt32(row["CreatedBy"])
  : 0;

                    movie.LastModifiedAt = row["LastModifiedAt"] != DBNull.Value
  ? Convert.ToDateTime(row["LastModifiedAt"])
  : DateTime.MinValue;

                    movie.LastModifiedBy = row["LastModifiedBy"] != DBNull.Value
   ? Convert.ToInt32(row["LastModifiedBy"])
   : 0;
                }

            }
            catch
            {
                throw;
            }
            return movie;
        }

        /*19 may Edit Movie*/
        public int UpdateMovie(MovieViewModel movie)
        {
            int resultIsUpdated = 0;

            try
            {
                DbCommand com = db.GetStoredProcCommand("UpdateMovie");
                db.AddInParameter(com, "@MovieID", DbType.Int32, movie.MovieID);
                db.AddInParameter(com, "@MovieName", DbType.String, !string.IsNullOrEmpty(movie.MovieName) ? (object)movie.MovieName : DBNull.Value);
                db.AddInParameter(com, "@Description", DbType.String, movie.Description);
                db.AddInParameter(com, "@MovieImage", DbType.String, movie.MovieImage);
                db.AddInParameter(com, "@Duration", DbType.Int32, movie.Duration);
                db.AddInParameter(com, "@Language", DbType.String, movie.Language);
                db.AddInParameter(com, "@Genre", DbType.String, movie.Genre);
                db.AddInParameter(com, "@ReleaseDate", DbType.DateTime, movie.ReleaseDate);
                db.AddInParameter(com, "@IsActive", DbType.Boolean, movie.IsActive);
                db.AddInParameter(com, "@LastModifiedBy", DbType.Int32, movie.LastModifiedBy);

                resultIsUpdated = db.ExecuteNonQuery(com);
            }

            catch
            {
                throw;
            }

            return resultIsUpdated;
        }

        /*19 may Delete Movie*/

        public int Deletemovie(int MovieId)
        {
            int resultIsDeleted = 0;

            try
            {
                DbCommand com = db.GetStoredProcCommand("DeleteMovie");
                db.AddInParameter(com, "@MovieId", DbType.Int32, MovieId);
                resultIsDeleted = db.ExecuteNonQuery(com);
            }
            catch
            {
                throw;
            }
            return resultIsDeleted;
        }
    }
}
